    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const startBtn = document.getElementById('startTest');
            const pingResult = document.getElementById('pingResult');
            const downloadResult = document.getElementById('downloadResult');
            const uploadResult = document.getElementById('uploadResult');
            const testStatus = document.getElementById('testStatus');
            const progressBar = document.getElementById('progressBar');
            const qualityValue = document.getElementById('qualityValue');
            const connectionType = document.getElementById('connectionType');
            
            // تحديد نوع الاتصال
            function detectConnectionType() {
                const connection = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
                if (connection) {
                    if (connection.type) {
                        return connection.type;
                    } else if (connection.effectiveType) {
                        return connection.effectiveType;
                    }
                }
                return 'غير معروف';
            }
            
            connectionType.textContent = detectConnectionType();
            
            startBtn.addEventListener('click', startSpeedTest);
            
            function startSpeedTest() {
                // تعطيل الزر أثناء الاختبار
                startBtn.disabled = true;
                startBtn.innerHTML = '<i class="fas fa-sync-alt fa-spin"></i> جاري الفحص...';
                
                // إعادة تعيين النتائج
                pingResult.textContent = '--';
                downloadResult.textContent = '--';
                uploadResult.textContent = '--';
                qualityValue.textContent = '--';
                
                // بدء عملية الفحص
                testStatus.innerHTML = '<i class="fas fa-cog fa-spin"></i> جاري التحضير للفحص...';
                progressBar.style.width = '10%';
                
                // محاكاة عملية فحص السرعة
                simulateSpeedTest();
            }
            
            function simulateSpeedTest() {
                // هذه محاكاة لفحص السرعة - في التطبيق الحقيقي، سنستخدم مكتبة أو API لفحص السرعة الفعلي
                
                // محاكاة قياس Ping
                setTimeout(() => {
                    const ping = Math.floor(Math.random() * 50) + 10;
                    pingResult.textContent = ping;
                    testStatus.innerHTML = '<i class="fas fa-cog fa-spin"></i> جاري قياس سرعة التحميل...';
                    progressBar.style.width = '40%';
                    
                    // محاكاة قياس Download
                    setTimeout(() => {
                        const download = (Math.random() * 90 + 10).toFixed(2);
                        downloadResult.textContent = download;
                        testStatus.innerHTML = '<i class="fas fa-cog fa-spin"></i> جاري قياس سرعة الرفع...';
                        progressBar.style.width = '70%';
                        
                        // محاكاة قياس Upload
                        setTimeout(() => {
                            const upload = (Math.random() * 40 + 5).toFixed(2);
                            uploadResult.textContent = upload;
                            testStatus.innerHTML = '<i class="fas fa-check-circle"></i> اكتمل الفحص بنجاح!';
                            progressBar.style.width = '100%';
                            
                            // تحديد جودة الاتصال بناء على النتائج
                            const speed = parseFloat(download);
                            if (speed < 10) {
                                qualityValue.textContent = 'ضعيفة';
                                qualityValue.style.color = '#dc3545';
                            } else if (speed < 30) {
                                qualityValue.textContent = 'متوسطة';
                                qualityValue.style.color = '#fd7e14';
                            } else if (speed < 60) {
                                qualityValue.textContent = 'جيدة';
                                qualityValue.style.color = '#20c997';
                            } else {
                                qualityValue.textContent = 'ممتازة';
                                qualityValue.style.color = '#0d6efd';
                            }
                            
                            // إعادة تمكين الزر
                            startBtn.disabled = false;
                            startBtn.innerHTML = '<i class="fas fa-redo"></i> بدء فحص جديد';
                        }, 2000);
                    }, 3000);
                }, 1500);
            }
        });
    </script>
